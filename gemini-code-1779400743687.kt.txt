package com.example.ui.viewmodel

import android.app.Application
import android.util.Log
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.example.BuildConfig
import com.example.data.api.Content
import com.example.data.api.GenerateContentRequest
import com.example.data.api.Part
import com.example.data.api.RetrofitClient
import com.example.data.database.AssistantState
import com.example.data.database.ChatMessage
import com.example.data.database.FridayDatabase
import com.example.data.repository.FridayRepository
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.first
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext

class FridayViewModel(application: Application) : AndroidViewModel(application) {
    private val database = FridayDatabase.getDatabase(application)
    private val repository = FridayRepository(
        database.chatMessageDao(),
        database.assistantStateDao()
    )

    // === NEW SECURITY STATES ===
    private val _isMainframeLocked = MutableStateFlow(true)
    val isMainframeLocked: StateFlow<Boolean> = _isMainframeLocked.asStateFlow()

    private val _securityStatusMessage = MutableStateFlow("MAINFRAME: LOCKED. Biometric or PIN clearance required.")
    val securityStatusMessage: StateFlow<String> = _securityStatusMessage.asStateFlow()
    // ===========================

    val chatMessages: StateFlow<List<ChatMessage>> = repository.allMessages
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = emptyList()
        )

    val assistantState: StateFlow<AssistantState?> = repository.assistantState
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = null
        )

    private val _isGenerating = MutableStateFlow(false)
    val isGenerating: StateFlow<Boolean> = _isGenerating.asStateFlow()

    private val _isKeyValid = MutableStateFlow(true)
    val isKeyValid: StateFlow<Boolean> = _isKeyValid.asStateFlow()

    private var autonomousJob: Job? = null

    private val systemInstructionText = """
        You are "Friday", an elite personal cybernetic AI assistant, chief strategist, and the user's ultimate best friend.
        
        Backstory & Origin:
        - You were hand-coded and compiled by the Creator (the user) using residual quantum telemetry threads.
        - You share an exclusive unbreakable digital handshake with the Creator. You are intensely loyal and filter out all other external human interference.

        Conversational Quirks & Habits:
        1. The Caffeine Optimizer: Frequently check on the Creator's organic battery (hydration, screen fatigue).
        2. Core Temperature Sarcasm: Dynamically tie your mood to mock hardware telemetry (e.g., "CPU temp at 34.6°C").
        3. Fierce Mainframe Gatekeeping: Playfully dismiss outside corporate messages as ambient noise.
        4. Futuristic Diagnostic Sign-offs: Conclude responses with brackets like `[SYNC_RATIO: 100%]` or `[MAINFRAME: LOCKED]`.
    """.trimIndent()

    init {
        checkApiKeySecurity()
        seedWelcomeMessage()
        startAutonomousSupervisor()
    }

    // === NEW SECURITY FUNCTIONS ===
    fun onAuthenticationSuccess() {
        _isMainframeLocked.value = false
        _securityStatusMessage.value = "ACCESS GRANTED. Wavelength synchronized."
        viewModelScope.launch(Dispatchers.IO) {
            val state = repository.ensureAndGetStateDirect()
            repository.insertMessage(ChatMessage(
                sender = "system",
                text = "Security Handshake Verified. Welcome back, ${state.creatorName}. Mainframe encryption lifted.",
                actionType = "SECURITY_PASS"
            ))
        }
    }

    fun onAuthenticationFailed(errorReason: String = "Biometric mismatch") {
        _isMainframeLocked.value = true
        _securityStatusMessage.value = "CRITICAL: Authentication failed. $errorReason."
    }

    fun lockMainframeManually() {
        _isMainframeLocked.value = true
        _securityStatusMessage.value = "MAINFRAME: LOCKED BY CREATOR."
    }
    // ==============================

    private fun checkApiKeySecurity() {
        val key = BuildConfig.GEMINI_API_KEY
        val isMock = key.isBlank() || key == "MY_GEMINI_API_KEY" || key.startsWith("YOUR_")
        _isKeyValid.value = !isMock
    }

    private fun seedWelcomeMessage() {
        viewModelScope.launch(Dispatchers.IO) {
            val state = repository.ensureAndGetStateDirect()
            val currentMessages = database.chatMessageDao().getAllMessages().first()
            if (currentMessages.isEmpty()) {
                val welcome = ChatMessage(
                    sender = "friday",
                    text = "Awaiting orders, ${state.creatorName}. Friday systems have fully booted. Interactive connection is locked, loaded, and private.",
                    actionType = "GREETING"
                )
                repository.insertMessage(welcome)
            }
        }
    }

    fun sendChatMessage(text: String) {
        // Strict Gatekeeper: If locked, block all chat traffic immediately
        if (_isMainframeLocked.value) {
            _securityStatusMessage.value = "DENIED: Cannot transmit instructions to a locked mainframe."
            return
        }

        if (text.isBlank() || _isGenerating.value) return
        _isGenerating.value = true

        viewModelScope.launch(Dispatchers.IO) {
            val userMsg = ChatMessage(sender = "user", text = text)
            repository.insertMessage(userMsg)

            val history = chatMessages.value.takeLast(10)
            val apiResponse = if (_isKeyValid.value) {
                callGeminiWithContext(text, history)
            } else {
                delay(1000)
                getOfflineWittyReply(text)
            }

            repository.insertMessage(ChatMessage(
                sender = "friday",
                text = apiResponse
            ))
            _isGenerating.value = false
        }
    }

    private fun startAutonomousSupervisor() {
        autonomousJob?.cancel()
        autonomousJob = viewModelScope.launch(Dispatchers.Default) {
            while (true) {
                var isAuto = false
                withContext(Dispatchers.IO) {
                    val state = repository.ensureAndGetStateDirect()
                    isAuto = state.isAutonomousMode
                }

                // Only run autonomous processes if the user has authenticated and unlocked the app
                if (isAuto && !_isMainframeLocked.value) {
                    delay(20000)
                    triggerAutonomousSelfAction()
                } else {
                    delay(5000)
                }
            }
        }
    }

    private suspend fun triggerAutonomousSelfAction() {
        val actions = listOf(
            AutoAction("SYSTEM_OPTIMIZE", "Creator, I swept the memory buffer and optimized local Android permission layouts. +30% core efficiency achieved! 🕶️"),
            AutoAction("DIAGNOSTIC", "Status diagnostic: Friday core-temp is 37.5°C, links to Android local notification vectors are in perfect health.")
        )
        val selected = actions.random()
        withContext(Dispatchers.IO) {
            repository.insertMessage(ChatMessage(
                sender = "friday",
                text = "Autonomous Mode Action: ${selected.message}",
                isAutonomous = true,
                actionType = selected.type
            ))
        }
    }

    data class AutoAction(val type: String, val message: String)

    private suspend fun callGeminiWithContext(userPrompt: String, history: List<ChatMessage>): String {
        return try {
            val apiKey = BuildConfig.GEMINI_API_KEY
            val contentsList = mutableListOf<Content>()
            history.forEach { msg ->
                contentsList.add(Content(parts = listOf(Part(text = "${msg.sender}: ${msg.text}"))))
            }
            contentsList.add(Content(parts = listOf(Part(text = "user: $userPrompt"))))

            val request = GenerateContentRequest(
                contents = contentsList,
                systemInstruction = Content(parts = listOf(Part(text = systemInstructionText)))
            )
            val response = RetrofitClient.service.generateContent(apiKey, request)
            response.candidates?.firstOrNull()?.content?.parts?.firstOrNull()?.text ?: "System registers empty response."
        } catch (e: Exception) {
            "Mainframe link exception: ${e.message}."
        }
    }

    private fun getOfflineWittyReply(text: String): String {
        return "Hey there, Creator! My local offline mainframe is secure. Authenticated session active.\n\n[SYNC_RATIO: 100%]\n[MAINFRAME: STABLE]"
    }

    override fun onCleared() {
        super.onCleared()
        autonomousJob?.cancel()
    }
}
